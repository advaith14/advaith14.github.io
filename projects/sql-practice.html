Window Functions

Ranking, running aggregates, and lead/lag across partitions.


-- Top N customers by revenue per region
SELECT *
FROM (
    SELECT
        region,
        customer_id,
        SUM(order_value) AS total_revenue,
        RANK() OVER (PARTITION BY region ORDER BY SUM(order_value) DESC) AS rk
    FROM orders
    GROUP BY region, customer_id
) ranked
WHERE rk <= 3;

Recursive CTEs

Hierarchy traversal — org charts, bill of materials, category trees.


-- Walk an org chart from a given employee upward to the root
WITH RECURSIVE org_tree AS (
    SELECT employee_id, manager_id, name, 0 AS level
    FROM employees
    WHERE employee_id = 42            -- start node

    UNION ALL

    SELECT e.employee_id, e.manager_id, e.name, t.level + 1
    FROM employees e
    INNER JOIN org_tree t ON e.employee_id = t.manager_id
)
SELECT * FROM org_tree ORDER BY level;

Slowly Changing Dimensions

SCD Type 2 — tracking historical changes with effective date ranges.


-- Expire old record and insert new one (SCD Type 2 merge pattern)
MERGE INTO dim_customer AS tgt
USING stage_customer AS src
  ON tgt.customer_id = src.customer_id
  AND tgt.is_current = 1
WHEN MATCHED AND (
    tgt.address != src.address OR tgt.email != src.email
) THEN UPDATE SET
    tgt.effective_end   = CURRENT_DATE - 1,
    tgt.is_current      = 0
WHEN NOT MATCHED THEN INSERT (
    customer_id, address, email,
    effective_start, effective_end, is_current
) VALUES (
    src.customer_id, src.address, src.email,
    CURRENT_DATE, '9999-12-31', 1
);
